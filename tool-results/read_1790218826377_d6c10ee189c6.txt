   302→/** On-load, staggered fade-up (hero). */
   303→function FadeUp({
   304→  delay = 0,
   305→  className,
   306→  children,
   307→}: {
   308→  delay?: number;
   309→  className?: string;
   310→  children: React.ReactNode;
   311→}) {
   312→  const reduced = useReducedMotion();
   313→  return (
   314→    <motion.div
   315→      className={className}
   316→      initial={reduced ? { opacity: 0 } : { opacity: 0, y: 24 }}
   317→      animate={{ opacity: 1, y: 0 }}
   318→      transition={{ duration: 0.7, delay, ease: 'easeOut' }}
   319→    >
   320→      {children}
   321→    </motion.div>
   322→  );
   323→}

   325→/** Scroll-into-view fade-up; plays once. */
   326→function Reveal({
   327→  delay = 0,
   328→  className,
   329→  children,
   330→}: {
   331→  delay?: number;
   332→  className?: string;
   333→  children: React.ReactNode;
   334→}) {
   335→  const reduced = useReducedMotion();
   336→  return (
   337→    <motion.div
   338→      className={className}
   339→      initial={reduced ? { opacity: 0 } : { opacity: 0, y: 18 }}
   340→      whileInView={{ opacity: 1, y: 0 }}
   341→      viewport={{ once: true, margin: '-64px' }}
   342→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
   343→    >
   344→      {children}
   345→    </motion.div>
   346→  );
   347→}
