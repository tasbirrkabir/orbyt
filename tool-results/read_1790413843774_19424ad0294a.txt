     5→/**
     6→ * The Command Center seeder. Builds three complete demo tenants with
     7→ * coherent, honest stories. Every block is idempotent: upserts keyed
     8→ * by natural ids (clientId, email, code, number) or count-guards, so
     9→ * re-running repairs and never duplicates.
    10→ *
    11→ * Honesty rules baked in: files travel by email (no fake download
    12→ * links), report numbers are labeled sample data, invoices describe
    13→ * real transfer methods, and no metric or health state is invented
    14→ * beyond the sample story the demo openly is.
    15→ */
    16→
    17→const DAY = 86_400_000;
    18→const HOUR = 3_600_000;
    19→const ago = (days: number, hours = 0) => new Date(Date.now() - days * DAY - hours * HOUR);
    20→const ahead = (days: number, hours = 0) => new Date(Date.now() + days * DAY + hours * HOUR);
    21→
    22→function parse<T>(json: string, fallback: T): T {
    23→  try {
    24→    return JSON.parse(json) as T;
    25→  } catch {
    26→    return fallback;
    27→  }
    28→}

    51→      tier: account.tier,
    52→      country: "BD",
    53→    },
    54→  });
    55→}

    57→async function ensureOwner(
    58→  account: (typeof DEMO_ACCOUNTS)[number],
    59→  organizationId: string,
    60→) {
    61→  const owner = await db.user.upsert({
    62→    where: { email: account.email },
    63→    update: { organizationId, role: "CLIENT_OWNER", clientId: account.clientId },
    64→    create: {
    65→      email: account.email,
    66→      name: account.name,
    67→      passwordHash: hashPassword(account.password),
    68→      clientId: account.clientId,
    69→      organizationId,
    70→      role: "CLIENT_OWNER",
    71→      activatedAt: ago(DEMO_AGE_DAYS),
    72→      // Created "a week ago" so the workspace shows a believable tenure.
    73→      createdAt: ago(DEMO_AGE_DAYS),
    74→    },
    75→  });
    76→  if (!owner.passwordHash) {
    77→    await db.user.update({
    78→      where: { id: owner.id },
    79→      data: { passwordHash: hashPassword(account.password), activatedAt: new Date() },
    80→    });
    81→  }
    82→  return owner;
    83→}

    85→async function ensureMembers(
    86→  members: MemberSeed[],
    87→  organizationId: string,
    88→  clientId: string,
    89→  password: string,
    90→) {
    91→  for (const member of members) {
    92→    await db.user.upsert({
    93→      where: { email: member.email },
    94→      update: { organizationId },
    95→      create: {
    96→        email: member.email,
    97→        name: member.name,
    98→        organizationId,
    99→        role: member.role,
   100→        // Invited members get a working password; viewers stay pending.
   101→        passwordHash: member.active ? hashPassword(password) : null,
   102→        activatedAt: member.active ? ago(DEMO_AGE_DAYS) : null,
   103→        // Pending members have no portal key of their own yet.
   104→        clientId: null,
   105→      },
   106→    });
   107→  }
   108→  void clientId;
   109→}

   111→async function ensureActivity(
   112→  organizationId: string,
   113→  events: {
   114→    type: string;
   115→    title: string;
   116→    actorName: string;
   117→    actorRole: string;
   118→    targetType?: string;
   119→    targetLabel?: string;
   120→    link?: string;
   121→    createdAt: Date;
   122→  }[],
   123→) {
   124→  const count = await db.activityEvent.count({ where: { organizationId } });
   125→  if (count > 0) return;
   126→  for (const event of events) {
   127→    await db.activityEvent.create({
   128→      data: {
   129→        organizationId,
   130→        type: event.type,
   131→        title: event.title,
   132→        actorName: event.actorName,
   133→        actorRole: event.actorRole,
   134→        targetType: event.targetType ?? "",
   135→        targetLabel: event.targetLabel ?? "",
   136→        link: event.link ?? null,
   137→        createdAt: event.createdAt,
   138→      },
   139→    });
   140→  }
   141→}
