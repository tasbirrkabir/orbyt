    70→  async function submitDecision(decision: "APPROVE" | "REQUEST_CHANGES" | "ACCEPTED" | "DECLINED") {
    71→    if (!decideItem || decideBusy) return;
    72→    const note = decideNote.trim();
    73→    if (decideItem.kind === "APPROVAL" && decision === "REQUEST_CHANGES" && note.length < 4) {
    74→      setDecideError("Say what should change — a comment is required.");
    75→      return;
    76→    }
    77→    if (decideItem.kind === "CHANGE_REQUEST" && decision === "DECLINED" && note.length < 4) {
    78→      setDecideError("Say why you are declining — a note is required.");
    79→      return;
    80→    }
    81→    setDecideBusy(true);
    82→    setDecideError(null);
    83→    try {
    84→      const url =
    85→        decideItem.kind === "APPROVAL"
    86→          ? `/api/portal/approvals/${decideItem.id.replace("approval-", "")}`
    87→          : `/api/portal/change-requests/${decideItem.id.replace("cr-", "")}`;
    88→      const body =
    89→        decideItem.kind === "APPROVAL"
    90→          ? { decision, comment: note }
    91→          : { decision, note: note || undefined };
    92→      const response = await fetch(url, {
    93→        method: "POST",
    94→        headers: { "Content-Type": "application/json" },
    95→        body: JSON.stringify(body),
    96→      });
    97→      if (!response.ok) {
    98→        const payload = (await response.json().catch(() => ({}))) as { error?: string };
    99→        setDecideError(payload.error ?? "That did not go through. Try again.");
   100→        return;
   101→      }
   102→      setDecideItem(null);
   103→      setDecideNote("");
   104→      refresh();
   105→    } catch {
   106→      setDecideError("Network trouble. Try again.");
   107→    } finally {
   108→      setDecideBusy(false);
   109→    }
   110→  }
